cmake_minimum_required(VERSION 3.16)
project(learn_transformer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(transformer src/main.cpp)
target_include_directories(transformer PRIVATE include)
if(MSVC)
  target_compile_options(transformer PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(transformer PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME correctness COMMAND transformer test)
